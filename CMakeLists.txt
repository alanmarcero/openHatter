cmake_minimum_required(VERSION 3.22)

project(OpenHatter VERSION 1.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(MSVC)
    add_compile_options(/utf-8)
endif()

option(OH_BUILD_PLUGIN "Build the plugin" ON)
option(OH_BUILD_TESTS "Build the unit tests" ON)

include(FetchContent)

add_library(openhatter_core STATIC
    Source/Core/HatSynth.cpp
    Source/Core/Effects.cpp
    Source/Core/HatEngine.cpp)
target_include_directories(openhatter_core PUBLIC Source/Core)
set_target_properties(openhatter_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(OH_BUILD_PLUGIN)
    FetchContent_Declare(juce
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.9
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(juce)

    juce_add_binary_data(HatterAssets SOURCES
        assets/openhatter-faceplate.png
        assets/mad-hatter.png
        assets/cyber-hatter-hero.png)
    set_target_properties(HatterAssets PROPERTIES POSITION_INDEPENDENT_CODE ON)

    juce_add_plugin(OpenHatter
        COMPANY_NAME "Alan Marcero"
        BUNDLE_ID "com.alanmarcero.openhatter"
        PRODUCT_NAME "openHatter"
        PLUGIN_MANUFACTURER_CODE Amrc
        PLUGIN_CODE Ohat
        FORMATS VST3 AU LV2 Standalone
        LV2URI "https://github.com/alanmarcero/openHatter"
        VST3_CATEGORIES Instrument Drum Synth
        AU_MAIN_TYPE kAudioUnitType_MusicDevice
        IS_SYNTH TRUE
        NEEDS_MIDI_INPUT TRUE
        NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS FALSE
        COPY_PLUGIN_AFTER_BUILD FALSE)

    target_sources(OpenHatter PRIVATE
        Source/Plugin/Parameters.cpp
        Source/Plugin/Presets.cpp
        Source/Plugin/EdmLookAndFeel.cpp
        Source/Plugin/Controls.cpp
        Source/Plugin/Panels.cpp
        Source/Plugin/PluginProcessor.cpp
        Source/Plugin/PluginEditor.cpp)

    target_include_directories(OpenHatter PRIVATE
        Source/Core
        Source/Plugin)

    target_compile_definitions(OpenHatter PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0)

    target_link_libraries(OpenHatter
        PRIVATE
            openhatter_core
            HatterAssets
            juce::juce_audio_processors
            juce::juce_audio_utils
            juce::juce_gui_basics
        PUBLIC
            juce::juce_recommended_config_flags
            juce::juce_recommended_lto_flags
            juce::juce_recommended_warning_flags)
endif()

if(OH_BUILD_TESTS)
    FetchContent_Declare(Catch2
        GIT_REPOSITORY https://github.com/catchorg/Catch2.git
        GIT_TAG v3.9.1
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(Catch2)

    add_executable(openhatter_tests
        Tests/HatSynthTests.cpp
        Tests/EffectsTests.cpp
        Tests/HatEngineTests.cpp)

    target_include_directories(openhatter_tests PRIVATE Tests)
    target_link_libraries(openhatter_tests PRIVATE openhatter_core Catch2::Catch2WithMain)

    enable_testing()
    include(Catch)
    catch_discover_tests(openhatter_tests)

    if(OH_BUILD_PLUGIN)
        add_executable(openhatter_plugin_tests Tests/PluginTests.cpp)
        target_include_directories(openhatter_plugin_tests PRIVATE Source/Plugin Source/Core)
        target_compile_definitions(openhatter_plugin_tests PRIVATE $<TARGET_PROPERTY:OpenHatter,COMPILE_DEFINITIONS>)
        target_link_libraries(openhatter_plugin_tests PRIVATE OpenHatter HatterAssets juce::juce_audio_utils Catch2::Catch2WithMain)
        catch_discover_tests(openhatter_plugin_tests)
    endif()
endif()
